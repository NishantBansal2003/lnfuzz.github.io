Returns a string, possibly empty, so callers guard with `with`. */ -}} {{- $url := "" -}} {{- with .url }}{{ $url = . }}{{ end -}} {{- if not $url }}{{ with .github }}{{ $url = printf "https://github.com/%s" . }}{{ end }}{{ end -}} {{- $url -}}
